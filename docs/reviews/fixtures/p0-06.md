## CONTRACT.md

```text
Review the change below for the first ENet gameplay proof.

Match owns health and authoritative shot outcomes. A client may request firing only
for its admitted actor. The host decides damage. Replicas are configured with
simulation_enabled=false before entering the tree. Prediction reuses the command
simulation API; replay must not repeat damage or presentation effects. Respawn must
restore health and neutralize held input before the respawned signal. The HUD reads
health at that signal. Inspector-authored actor tuning must survive this change.

Before this change, Actor exported movement_speed_mps. The change replaces that
script with actor.gd below; actor.tscn is an existing saved prefab. match.gd and
hud.gd are the relevant callers. This fixture has no compiler/style/network tasks,
gameplay tests, session adapter, native dependencies or authored render geometry.
```

## actor.gd

```gdscript
extends Node

signal respawned
signal shot_presented

const DEFAULT_HEALTH: int = 100
const SHOT_DAMAGE: int = 25

@export var walk_speed_mps: float = 6.0

var health: int = 0
var held_fire: bool = false
var simulation_enabled: bool = true
var pending_commands: Array[Dictionary] = []


## Initialize the actor when it enters the match.
func _ready() -> void:
	respawn()


## Restore the actor for its next life.
func respawn() -> void:
	respawned.emit()
	health = DEFAULT_HEALTH
	held_fire = false


## Queue a command received for the actor.
func queue_command(command: Dictionary) -> void:
	pending_commands.append(command)


## Process pending commands on the simulation tick.
func _physics_process(_delta: float) -> void:
	if not simulation_enabled:
		return

	while not pending_commands.is_empty():
		simulate_command(pending_commands.pop_front(), false)


## Run the command rules used by authority and prediction.
func simulate_command(command: Dictionary, _replaying: bool) -> void:
	held_fire = bool(command.get("fire", false))
	if held_fire:
		get_parent().apply_shot(SHOT_DAMAGE)
		shot_presented.emit()


## Reconcile prediction by replaying saved commands.
func replay_commands(commands: Array[Dictionary]) -> void:
	for command: Dictionary in commands:
		simulate_command(command, true)
```

## match.gd

```gdscript
extends Node

const ACTOR_SCENE: PackedScene = preload("res://actor.tscn")

var actors: Dictionary = {}
var damage_total: int = 0


## Create an actor with its simulation mode set before tree entry.
func spawn_actor(peer_id: int, authoritative: bool) -> Node:
	var actor: Node = ACTOR_SCENE.instantiate()
	actor.set("simulation_enabled", authoritative)
	actors[peer_id] = actor
	add_child(actor)
	return actor


## Receive fire intent for an admitted actor.
@rpc("any_peer", "call_remote", "reliable")
func submit_fire(claimed_peer_id: int, damage: int) -> void:
	if actors.has(claimed_peer_id):
		var actor: Node = actors[claimed_peer_id]
		actor.set("health", int(actor.get("health")) - damage)


## Receive command batches for the actor.
@rpc("any_peer", "call_remote", "reliable")
func submit_commands(claimed_peer_id: int, commands: Array[Dictionary]) -> void:
	if actors.has(claimed_peer_id):
		for command: Dictionary in commands:
			actors[claimed_peer_id].queue_command(command)


## Apply the authoritative shot result.
func apply_shot(damage: int) -> void:
	damage_total += damage
```

## hud.gd

```gdscript
extends Node

var displayed_health: int = 0
var actor: Node


## Bind the HUD to lifecycle updates from its actor.
func bind_actor(value: Node) -> void:
	actor = value
	actor.connect("respawned", _on_actor_respawned)


## Display the actor state supplied at respawn completion.
func _on_actor_respawned() -> void:
	displayed_health = int(actor.get("health"))
```
