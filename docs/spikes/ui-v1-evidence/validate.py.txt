"""Reproducible documentation-asset audit; not a project/runtime tool.

From repository root, with a fresh external output path:
python docs/spikes/ui-v1-evidence/validate.py.txt --output C:/tmp/ft/lanes/ui-mockups/audit-01/static-validation.json
"""
import argparse
import base64
import hashlib
import json
from pathlib import Path
import re
import xml.etree.ElementTree as ET

from PIL import Image

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('--output', type=Path, required=True)
args = parser.parse_args()
root = Path.cwd()
assert not args.output.resolve().is_relative_to(root), 'Output must be outside checkout'
assert not args.output.exists(), 'Output must be fresh'
folder = root / 'docs/concepts/ui-v1'
evidence = root / 'docs/spikes/ui-v1-evidence'
expected = [
    '01-foot-compact', '02-foot-balanced', '03-foot-wide', '04-foot-art-context',
    '05-foot-damage', '06-car-moving', '07-car-stopped', '08-minimap-directions',
    '09-main-menu', '10-host', '11-join-steam', '12-join-direct', '13-lobby',
    '14-pause', '15-audio-settings', '16-host-lost', '17-join-error', '18-join-loading',
]
assert sorted(p.stem for p in folder.glob('*.svg')) == expected
assert sorted(p.stem for p in folder.glob('*.png')) == expected
sources = [
    'docs/spikes/s06-windows-evidence/captures/camera-42/static.png',
    'docs/spikes/s05-windows-draw-evidence/client/burst.png',
    'docs/concepts/p0-04/i-perspective-high-rise.png',
]
sha = lambda data: hashlib.sha256(data).hexdigest()
source_hashes = {name: sha((root / name).read_bytes()) for name in sources}
ns = {'s': 'http://www.w3.org/2000/svg'}
manifest = json.loads((evidence / 'renders/render-results.json').read_text())
rows = []
for name in expected:
    svg = folder / (name + '.svg')
    png = svg.with_suffix('.png')
    tree = ET.fromstring(svg.read_bytes())
    assert tree.attrib['width'] == '1280' and tree.attrib['height'] == '800'
    assert tree.attrib['viewBox'] == '0 0 1280 800'
    assert tree.find('s:title', ns) is not None
    assert tree.find('s:desc', ns) is not None
    assert not tree.findall('.//s:script', ns)
    assert not tree.findall('.//s:foreignObject', ns)
    embeds = []
    for node in tree.iter():
        for key, value in node.attrib.items():
            assert not key.startswith('on'), (svg, key)
            if key == 'href' or key.endswith('}href'):
                assert value.startswith('data:image/png;base64,') or value.startswith('#')
        if node.tag == '{http://www.w3.org/2000/svg}image':
            data = base64.b64decode(node.attrib['href'].split(',', 1)[1], validate=True)
            digest = sha(data)
            assert digest in source_hashes.values()
            embeds.append(next(p for p, h in source_hashes.items() if h == digest))
    with Image.open(png) as image:
        assert image.size == (1280, 800)
        image.verify()
    original = next(r for r in manifest if r['svg'].endswith(name + '.svg'))
    assert original['ok'] and original['returncode'] == 0
    assert original['svg_sha256'] == sha(svg.read_bytes())
    assert original['png_sha256'] == sha(png.read_bytes())
    rows.append({'screen': name, 'svg_sha256': original['svg_sha256'],
                 'png_sha256': original['png_sha256'], 'embedded_sources': embeds,
                 'size': [1280, 800], 'ok': True})

# Independent expected state labels, not strings copied from a generator output list.
def texts(name):
    tree = ET.fromstring((folder / (name + '.svg')).read_bytes())
    return [node.text for node in tree.findall('.//s:text', ns)]

for name in expected[:3]:
    values = texts(name)
    assert 'SMG' in values and '24 / 30' in values and 'enter car' in values
    assert {'Mika', 'Jules', 'Nico'} <= set(values)
assert 'LOW HEALTH' in texts('05-foot-damage')
assert 'exit car' not in texts('06-car-moving')
assert 'Stop the car to exit' in texts('06-car-moving')
assert 'exit car' in texts('07-car-stopped')
for name in ['06-car-moving', '07-car-stopped']:
    assert 'SMG' not in texts(name) and 'Car health' in texts(name)
assert 'Return to main menu' in texts('16-host-lost')
assert {'Master', 'Music', 'SFX', 'MUTE ALL'} <= set(texts('15-audio-settings'))
assert 'Cancel' in texts('18-join-loading')

links = []
for doc in [folder / 'README.md', folder / 'review.html', evidence / 'README.md',
            evidence / 'initial-attempt/README.md']:
    content = doc.read_text(encoding='utf-8')
    targets = re.findall(r'(?:href|src)="([^"]+)"', content)
    targets += re.findall(r'\]\(([^)]+)\)', content)
    for target in targets:
        if target.startswith(('http:', 'https:', '#', 'data:')):
            continue
        local = target.split('#', 1)[0]
        assert (doc.parent / local).exists(), (doc, local)
        links.append({'from': doc.relative_to(root).as_posix(), 'to': local})

files = [p for directory in [folder, evidence] for p in directory.rglob('*') if p.is_file()]
assert all(p.stat().st_size < 5_000_000 for p in files)
assert not any(p.suffix in {'.gd', '.tscn', '.tres', '.py'} for p in files)

# Solid-token contrast only, not a device/accessibility certification.
def luminance(color):
    channels = [int(color[i:i+2], 16) / 255 for i in [1, 3, 5]]
    linear = [x / 12.92 if x <= 0.04045 else ((x + 0.055) / 1.055) ** 2.4 for x in channels]
    return sum(x * w for x, w in zip(linear, [0.2126, 0.7152, 0.0722]))

contrast = {}
for name, foreground, background in [
    ('ivory/panel', '#F6F1DC', '#102F3A'), ('muted/panel', '#AEC3C6', '#102F3A'),
    ('coral/panel', '#FF725D', '#102F3A'), ('ink/coral', '#0C222B', '#FF725D'),
    ('amber/panel', '#FFC05A', '#102F3A')]:
    light, dark = sorted([luminance(foreground), luminance(background)], reverse=True)
    contrast[name] = round((light + .05) / (dark + .05), 2)
    assert contrast[name] >= 4.5
result = {'ok': True, 'screens': rows, 'source_sha256': source_hashes,
          'screen_count': len(rows), 'links_checked': len(links),
          'delivered_files_checked_for_size_and_scope': len(files),
          'largest_file_bytes': max(p.stat().st_size for p in files),
          'contrast_ratios': contrast,
          'limitations': ['Solid-token contrast only; no physical Deck or accessibility certification.',
                         'State text assertions do not exercise actual gameplay or menu behaviour.']}
args.output.parent.mkdir(parents=True, exist_ok=True)
args.output.write_text(json.dumps(result, indent=2) + '\n', encoding='utf-8')
print(json.dumps({key: value for key, value in result.items() if key not in ['screens', 'source_sha256']}))
