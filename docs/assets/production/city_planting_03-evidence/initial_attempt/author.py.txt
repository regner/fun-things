# Crown coordinates are authored asymmetrically. Broad three-lobe modulation is
# deliberately below leaf-scale; closed tips keep export triangulation unambiguous.
def crown(name,cx,cy,cz,rx,ry,rz,phase,material):
    n=32; rings=15
    vs=[(cx,cy,cz-rz)]
    for j in range(1,rings+1):
        t=-math.pi/2+math.pi*j/(rings+1)
        for i in range(n):
            a=2*math.pi*i/n
            wobble=1+.055*math.cos(3*a+phase)*math.cos(t)**2+.025*math.sin(2*a-phase)*math.cos(t)
            # Wide shoulders, gently compressed apex rather than a perfect ball.
            r=math.cos(t)**.84
            vs.append((cx+rx*r*math.cos(a)*wobble,cy+ry*r*math.sin(a)*wobble,
                       cz+rz*math.sin(t)+.013*math.cos(2*a+phase)*math.cos(t)**2))
    vs.append((cx,cy,cz+rz)); top=len(vs)-1
    fs=[(0,1+(i+1)%n,1+i) for i in range(n)]
    for j in range(rings-1):
        for i in range(n):
            a=1+j*n+i; b=1+j*n+(i+1)%n
            fs.append((a,b,b+n,a+n))
    fs.extend((1+(rings-1)*n+i,1+(rings-1)*n+(i+1)%n,top) for i in range(n))
    return mesh(name,vs,fs,material)
