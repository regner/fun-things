# A profile, not a scaled primitive: tucked base, broad low shoulder, offset dome.
PROFILE=[(0,0),(.065,.40),(.15,.68),(.27,.90),(.40,1),(.54,.99),(.68,.91),(.80,.76),(.90,.56),(.965,.30),(1,0)]
def crown(name,center,radii,phase,mat,col,root):
    cx,cy,cz=center; rx,ry,h=radii; n=40
    vs=[(cx-.05,cy,cz)]
    for z,r in PROFILE[1:-1]:
        for i in range(n):
            a=2*math.pi*i/n
            wave=1+.065*math.cos(3*a+phase)+.025*math.sin(5*a-phase)
            lean=.11*math.sin(math.pi*z)
            vs.append((cx+rx*r*math.cos(a)*wave+lean,
                       cy+ry*r*math.sin(a)*wave+.045*math.sin(2*a+phase)*r,
                       cz+h*z+.035*math.cos(2*a+phase)*math.sin(math.pi*z)))
    vs.append((cx+.035,cy,cz+h)); top=len(vs)-1; rings=len(PROFILE)-2
    fs=[(0,1+(i+1)%n,1+i) for i in range(n)]
    for j in range(rings-1):
        for i in range(n):
            a=1+j*n+i; b=1+j*n+(i+1)%n; fs.append((a,b,b+n,a+n))
    fs.extend((1+(rings-1)*n+i,1+(rings-1)*n+(i+1)%n,top) for i in range(n))
    return mesh(name,vs,fs,mat,col,root)
def wood(name,paths,col,root):
    vs=[]; fs=[]; n=16
    for path in paths:
        start=len(vs)
        for j,(x,y,z,r) in enumerate(path):
            for i in range(n):
                a=2*math.pi*i/n; flare=1+.09*math.cos(5*a+.3)*max(0,1-z/.5)
                vs.append((x+r*math.cos(a)*flare,y+r*math.sin(a)*flare,z))
        fs.append(tuple(start+i for i in reversed(range(n))))
        for j in range(len(path)-1):
            for i in range(n):
                a=start+j*n+i; b=start+j*n+(i+1)%n; fs.append((a,b,b+n,a+n))
        fs.append(tuple(start+(len(path)-1)*n+i for i in range(n)))
    o=mesh(name,vs,fs,bark,col,root)
    # Unite branch collars locally; the finished editable mesh retains a smooth fork.
    bpy.context.view_layer.objects.active=o; o.select_set(True)
    rem=o.modifiers.new('unified_branch_collars','REMESH'); rem.mode='VOXEL'; rem.voxel_size=.025; rem.use_smooth_shade=True
    bpy.ops.object.modifier_apply(modifier=rem.name)
    smooth=o.modifiers.new('soft_wood','SMOOTH'); smooth.factor=.6; smooth.iterations=3
    bpy.ops.object.modifier_apply(modifier=smooth.name)
    dec=o.modifiers.new('restrained_wood_density','DECIMATE'); dec.ratio=.35
    bpy.ops.object.modifier_apply(modifier=dec.name)
    # Ground datum exact after voxel union; original flare remains below 0.23 m radius.
    low=min(v.co.z for v in o.data.vertices)
    for v in o.data.vertices:v.co.z-=low
    o.select_set(False)
    return o
